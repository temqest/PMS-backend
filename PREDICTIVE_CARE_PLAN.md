# Predictive Care Feature — Build Plan
> Node.js · Express · MongoDB  
> Connects to existing `HealthRecord`, `Patient`, `Appointment`, and `PrescriptionInvoice` collections.

---

## Overview

This plan adds a self-contained **Predictive Care** module to your existing PMS backend. It reads from your current health record collections, builds its own MongoDB collections for storing computed risk data, and exposes a clean API for your frontend to render graphs, scores, and alerts.

### What gets built
| Feature | What it does |
|---|---|
| Lab trend detection | Tracks value changes across lab results over time per patient |
| Chronic disease risk scoring | Computes a risk score from vitals + diagnoses |
| Vaccination gap detection | Flags overdue or incomplete vaccination series |
| Medication adherence monitoring | Detects refill gaps and drop-offs |
| No-show / appointment risk | Predicts likelihood of skipping upcoming appointments |
| Readmission risk | Composite score from visit frequency + critical labs |
| Analytics API | Chart-ready endpoints for the frontend dashboard |

### 2.1 Lab Trend Service

Reads all `Lab Result` health records for a patient and computes trend direction.

```js
// src/api/v1/predictive-care/services/labTrend.service.js
const HealthRecord = require('../../health-records/healthRecord.model');
const LabTrend = require('../models/labTrend.model');
const CareAlert = require('../models/careAlert.model');

// Called after any new Lab Result record is saved
const computeLabTrendsForPatient = async (patient_id, patient_name) => {
  // Fetch all lab result records for this patient, oldest first
  const labRecords = await HealthRecord.find({
    patient_id,
    record_type: 'Lab Result',
    save_state: 'final',
  }).sort({ record_date: 1 });

  if (!labRecords.length) return;

  // Group by test name
  const byTest = {};
  for (const record of labRecords) {
    const name = record.details?.labTestName;
    if (!name) continue;
    if (!byTest[name]) byTest[name] = [];

    // Parse numeric value from string like "182 mg/dL" or "182"
    const rawStr = record.details?.labResultValue || '';
    const numericValue = parseFloat(rawStr);
    if (isNaN(numericValue)) continue;

    byTest[name].push({
      record_id: record.record_id,
      value: numericValue,
      raw_value_string: rawStr,
      unit: record.details?.labUnit || '',
      status: record.details?.labStatus || 'Normal',
      recorded_at: record.record_date,
    });
  }

  for (const [testName, points] of Object.entries(byTest)) {
    const trend = _computeTrend(points);
    const consecutiveAbnormal = _countConsecutiveAbnormal(points);
    const lastPoint = points[points.length - 1];

    await LabTrend.findOneAndUpdate(
      { patient_id, test_name: testName },
      {
        patient_id,
        test_name: testName,
        data_points: points,
        trend_direction: trend.direction,
        trend_severity: trend.severity,
        consecutive_abnormal_count: consecutiveAbnormal,
        last_value: lastPoint.value,
        last_status: lastPoint.status,
        last_recorded_at: lastPoint.recorded_at,
        alert_triggered: consecutiveAbnormal >= 2 || trend.severity === 'Severe',
      },
      { upsert: true, new: true }
    );

    // Fire alert if trend is worsening with 2+ abnormal readings
    if (consecutiveAbnormal >= 2) {
      await _upsertLabAlert(patient_id, patient_name, testName, lastPoint, trend, consecutiveAbnormal);
    }
  }
};

const _computeTrend = (points) => {
  if (points.length < 2) return { direction: 'Insufficient data', severity: 'None' };

  // Use last 3 data points for trend
  const recent = points.slice(-3);
  const first = recent[0].value;
  const last = recent[recent.length - 1].value;
  const changePct = ((last - first) / (first || 1)) * 100;

  const abnormalCount = recent.filter(p => p.status !== 'Normal').length;

  let direction = 'Stable';
  if (changePct > 15) direction = 'Worsening';
  else if (changePct < -15) direction = 'Improving';

  let severity = 'None';
  if (abnormalCount === recent.length && direction === 'Worsening') severity = 'Severe';
  else if (abnormalCount >= 2) severity = 'Moderate';
  else if (abnormalCount === 1) severity = 'Mild';

  return { direction, severity };
};

const _countConsecutiveAbnormal = (points) => {
  let count = 0;
  for (let i = points.length - 1; i >= 0; i--) {
    if (points[i].status !== 'Normal') count++;
    else break;
  }
  return count;
};

const _upsertLabAlert = async (patient_id, patient_name, testName, lastPoint, trend, consecutiveCount) => {
  const severity = consecutiveCount >= 3 ? 'Critical' : 'Warning';
  await CareAlert.findOneAndUpdate(
    { patient_id, alert_type: 'LAB_TREND', 'metadata.test_name': testName, is_resolved: false },
    {
      patient_id,
      patient_name,
      alert_type: 'LAB_TREND',
      severity,
      title: `${testName} — ${trend.direction}`,
      message: `${testName} has been ${lastPoint.status.toLowerCase()} for ${consecutiveCount} consecutive readings. Last value: ${lastPoint.raw_value_string}.`,
      metadata: {
        test_name: testName,
        trend_direction: trend.direction,
        last_value: lastPoint.raw_value_string,
        consecutive_abnormal: consecutiveCount,
      },
      triggered_at: new Date(),
    },
    { upsert: true, new: true }
  );
};

module.exports = { computeLabTrendsForPatient };
```

### 2.2 Risk Score Service

Computes chronic disease and readmission risk from Visit records.

```js
// src/api/v1/predictive-care/services/riskScore.service.js
const HealthRecord = require('../../health-records/healthRecord.model');
const Appointment = require('../../appointments/appointment.model');
const PatientRiskProfile = require('../models/patientRiskProfile.model');
const CareAlert = require('../models/careAlert.model');
const dayjs = require('dayjs');

const HIGH_RISK_DIAGNOSES = [
  'hypertension', 'type 2 diabetes', 'chronic', 'heart failure',
  'copd', 'obesity', 'renal', 'kidney', 'coronary'
];

const computeRiskProfileForPatient = async (patient) => {
  const { patient_id, first_name, last_name } = patient;
  const patient_name = `${first_name} ${last_name}`;

  const [visitRecords, labRecords, appointments, allRecords] = await Promise.all([
    HealthRecord.find({ patient_id, record_type: 'Visit', save_state: 'final' }).sort({ record_date: -1 }),
    HealthRecord.find({ patient_id, record_type: 'Lab Result', save_state: 'final' }).sort({ record_date: -1 }),
    Appointment.find({ patient_id }).sort({ scheduled_at: -1 }),
    HealthRecord.countDocuments({ patient_id }),
  ]);

  const chronicScore = _computeChronicDiseaseScore(visitRecords, labRecords);
  const readmissionScore = _computeReadmissionScore(visitRecords);
  const noShowScore = _computeNoShowScore(appointments);
  const hasCriticalLabs = labRecords.some(r => r.details?.labStatus === 'Critical');

  // Weighted composite: chronic 40%, readmission 30%, no-show 15%, adherence placeholder 15%
  const overallScore = Math.round(
    chronicScore * 0.40 +
    readmissionScore * 0.30 +
    noShowScore * 0.15
  );

  const riskLevel = overallScore >= 75 ? 'Critical'
    : overallScore >= 50 ? 'High'
    : overallScore >= 25 ? 'Moderate'
    : 'Low';

  const profile = await PatientRiskProfile.findOneAndUpdate(
    { patient_id },
    {
      patient_id,
      patient_name,
      overall_risk_score: overallScore,
      overall_risk_level: riskLevel,
      chronic_disease_risk: chronicScore,
      readmission_risk: readmissionScore,
      no_show_risk: noShowScore,
      has_critical_labs: hasCriticalLabs,
      last_computed_at: new Date(),
      record_count_at_last_compute: allRecords,
    },
    { upsert: true, new: true }
  );

  // Fire alert for high overall risk
  if (overallScore >= 50) {
    await CareAlert.findOneAndUpdate(
      { patient_id, alert_type: 'CHRONIC_RISK', is_resolved: false },
      {
        patient_id,
        patient_name,
        alert_type: 'CHRONIC_RISK',
        severity: riskLevel === 'Critical' ? 'Critical' : 'Warning',
        title: `${riskLevel} chronic disease risk`,
        message: `Patient has a risk score of ${overallScore}/100 based on vitals, diagnoses, and visit history.`,
        metadata: { overall_score: overallScore, chronic_score: chronicScore, readmission_score: readmissionScore },
        triggered_at: new Date(),
      },
      { upsert: true, new: true }
    );
  }

  return profile;
};

const _computeChronicDiseaseScore = (visitRecords, labRecords) => {
  let score = 0;

  // Points for high-risk diagnoses in visit history
  const diagnoses = visitRecords.map(r => (r.details?.visitAssessment || '').toLowerCase());
  for (const d of diagnoses) {
    for (const keyword of HIGH_RISK_DIAGNOSES) {
      if (d.includes(keyword)) { score += 10; break; }
    }
  }

  // Points for abnormal/critical labs
  const abnormalLabs = labRecords.filter(r => r.details?.labStatus !== 'Normal');
  score += abnormalLabs.length * 8;

  // Points for high BP across visits
  const highBpVisits = visitRecords.filter(r => {
    const systolic = parseInt(r.details?.visitBpSystolic || '0');
    return systolic >= 140;
  });
  score += highBpVisits.length * 6;

  return Math.min(score, 100);
};

const _computeReadmissionScore = (visitRecords) => {
  // Count urgent/unplanned visits in the last 90 days
  const cutoff = dayjs().subtract(90, 'day').toDate();
  const recentUrgent = visitRecords.filter(r =>
    new Date(r.record_date) >= cutoff &&
    (r.details?.visitType === 'Urgent' || r.details?.visitType === 'Follow-up')
  );
  return Math.min(recentUrgent.length * 20, 100);
};

const _computeNoShowScore = (appointments) => {
  if (!appointments.length) return 0;
  const completed = appointments.filter(a => a.status === 'Completed').length;
  const cancelled = appointments.filter(a => a.status === 'Pending').length;
  const total = appointments.length;
  // Higher ratio of non-completed = higher risk
  const incompletePct = ((total - completed) / total) * 100;
  return Math.min(Math.round(incompletePct * 0.8 + cancelled * 5), 100);
};

// Run for all patients (nightly job / manual trigger)
const computeAllPatientRiskProfiles = async () => {
  const Patient = require('../../patients/patient.model');
  const patients = await Patient.find({ status: 'active' });
  console.log(`[PredictiveCare] Computing risk for ${patients.length} patients...`);
  for (const patient of patients) {
    await computeRiskProfileForPatient(patient);
  }
  console.log('[PredictiveCare] Done.');
};

module.exports = { computeRiskProfileForPatient, computeAllPatientRiskProfiles };
```

### 2.3 Vaccination Gap Service

```js
// src/api/v1/predictive-care/services/vaccinationGap.service.js
const HealthRecord = require('../../health-records/healthRecord.model');
const CareAlert = require('../models/careAlert.model');
const dayjs = require('dayjs');

const checkVaccinationGapsForPatient = async (patient_id, patient_name) => {
  const vaccinationRecords = await HealthRecord.find({
    patient_id,
    record_type: 'Vaccination',
    save_state: 'final',
  }).sort({ record_date: -1 });

  const today = dayjs();
  const alerts = [];

  for (const record of vaccinationRecords) {
    const { vaccinationName, vaccinationSeriesComplete, vaccinationNextDoseDue } = record.details || {};

    // Series incomplete and next dose date has passed
    if (!vaccinationSeriesComplete && vaccinationNextDoseDue) {
      const dueDate = dayjs(vaccinationNextDoseDue);
      const daysOverdue = today.diff(dueDate, 'day');

      if (daysOverdue > 0) {
        alerts.push({
          name: vaccinationName,
          daysOverdue,
          nextDoseDue: vaccinationNextDoseDue,
        });

        await CareAlert.findOneAndUpdate(
          {
            patient_id,
            alert_type: 'VACCINATION_GAP',
            'metadata.vaccine_name': vaccinationName,
            is_resolved: false,
          },
          {
            patient_id,
            patient_name,
            alert_type: 'VACCINATION_GAP',
            severity: daysOverdue > 90 ? 'Critical' : daysOverdue > 30 ? 'Warning' : 'Info',
            title: `${vaccinationName} dose overdue`,
            message: `Next dose of ${vaccinationName} was due ${daysOverdue} day(s) ago (${vaccinationNextDoseDue}). Series is incomplete.`,
            metadata: {
              vaccine_name: vaccinationName,
              days_overdue: daysOverdue,
              next_dose_due: vaccinationNextDoseDue,
            },
            triggered_at: new Date(),
          },
          { upsert: true, new: true }
        );
      }
    }
  }

  // Update the profile flag
  const PatientRiskProfile = require('../models/patientRiskProfile.model');
  await PatientRiskProfile.findOneAndUpdate(
    { patient_id },
    { has_overdue_vaccinations: alerts.length > 0 },
    { upsert: true }
  );

  return alerts;
};

module.exports = { checkVaccinationGapsForPatient };
```

### 2.4 Adherence Service

```js
// src/api/v1/predictive-care/services/adherence.service.js
const HealthRecord = require('../../health-records/healthRecord.model');
const PrescriptionInvoice = require('../../prescription-invoices/prescriptionInvoice.model');
const AdherenceRecord = require('../models/adherenceRecord.model');
const CareAlert = require('../models/careAlert.model');
const PatientRiskProfile = require('../models/patientRiskProfile.model');
const dayjs = require('dayjs');

const ADHERENCE_GAP_THRESHOLD_DAYS = 14; // flag if >14 days without a refill

const computeAdherenceForPatient = async (patient_id, patient_name) => {
  const prescriptionRecords = await HealthRecord.find({
    patient_id,
    record_type: 'Prescription',
    save_state: 'final',
  }).sort({ record_date: 1 });

  const invoices = await PrescriptionInvoice.find({ patient_id });

  // Build a map of confirmed fills from invoices: medicineName → fill dates
  const fillMap = {};
  for (const invoice of invoices) {
    for (const item of invoice.items || []) {
      if (!fillMap[item.medicineName]) fillMap[item.medicineName] = [];
      fillMap[item.medicineName].push(invoice.invoice_date);
    }
  }

  // Group prescription records by medicine name
  const byMedicine = {};
  for (const record of prescriptionRecords) {
    const name = record.details?.medicationName;
    if (!name) continue;
    if (!byMedicine[name]) byMedicine[name] = [];
    byMedicine[name].push({
      record_id: record.record_id,
      start_date: record.details?.prescriptionStartDate,
      end_date: record.details?.prescriptionEndDate,
      quantity: parseInt(record.details?.prescriptionQuantity || '0'),
      refills: parseInt(record.details?.prescriptionRefills || '0'),
    });
  }

  let overallAdherenceIssues = false;

  for (const [medicineName, windows] of Object.entries(byMedicine)) {
    const gaps = _findCoverageGaps(windows);
    const confirmedFills = (fillMap[medicineName] || []).map(d => ({ fill_date: d }));
    const significantGaps = gaps.filter(g => g.gap_days > ADHERENCE_GAP_THRESHOLD_DAYS);
    const longestGap = gaps.reduce((max, g) => Math.max(max, g.gap_days), 0);
    const adherenceScore = Math.max(0, 100 - significantGaps.length * 25);
    const status = adherenceScore >= 75 ? 'Adherent' : adherenceScore >= 40 ? 'Partial' : 'Non-adherent';

    await AdherenceRecord.findOneAndUpdate(
      { patient_id, medicine_name: medicineName },
      {
        patient_id,
        medicine_name: medicineName,
        prescription_windows: windows,
        coverage_gaps: gaps,
        confirmed_fills: confirmedFills,
        adherence_score: adherenceScore,
        status,
        longest_gap_days: longestGap,
        last_assessed_at: new Date(),
      },
      { upsert: true, new: true }
    );

    if (status !== 'Adherent') {
      overallAdherenceIssues = true;
      await CareAlert.findOneAndUpdate(
        { patient_id, alert_type: 'ADHERENCE_GAP', 'metadata.medicine_name': medicineName, is_resolved: false },
        {
          patient_id,
          patient_name,
          alert_type: 'ADHERENCE_GAP',
          severity: status === 'Non-adherent' ? 'Warning' : 'Info',
          title: `Adherence gap — ${medicineName}`,
          message: `Patient shows ${status.toLowerCase()} adherence to ${medicineName}. Longest gap: ${longestGap} days.`,
          metadata: { medicine_name: medicineName, adherence_score: adherenceScore, longest_gap_days: longestGap, status },
          triggered_at: new Date(),
        },
        { upsert: true, new: true }
      );
    }
  }

  await PatientRiskProfile.findOneAndUpdate(
    { patient_id },
    { has_adherence_gaps: overallAdherenceIssues },
    { upsert: true }
  );
};

const _findCoverageGaps = (windows) => {
  const gaps = [];
  for (let i = 0; i < windows.length - 1; i++) {
    const endA = dayjs(windows[i].end_date);
    const startB = dayjs(windows[i + 1].start_date);
    const gapDays = startB.diff(endA, 'day');
    if (gapDays > 0) {
      gaps.push({ gap_start: endA.toDate(), gap_end: startB.toDate(), gap_days: gapDays });
    }
  }
  return gaps;
};

module.exports = { computeAdherenceForPatient };
```

```js
// src/api/v1/predictive-care/services/analytics.service.js
const PatientRiskProfile = require('../models/patientRiskProfile.model');
const CareAlert = require('../models/careAlert.model');
const LabTrend = require('../models/labTrend.model');
const AdherenceRecord = require('../models/adherenceRecord.model');

// Population-level summary for the main dashboard
const getDashboardSummary = async () => {
  const [riskDistribution, alertCounts, topHighRisk] = await Promise.all([
    PatientRiskProfile.aggregate([
      { $group: { _id: '$overall_risk_level', count: { $sum: 1 } } }
    ]),
    CareAlert.aggregate([
      { $match: { is_resolved: false } },
      { $group: { _id: '$alert_type', count: { $sum: 1 } } }
    ]),
    PatientRiskProfile.find({ overall_risk_level: { $in: ['High', 'Critical'] } })
      .sort({ overall_risk_score: -1 })
      .limit(10)
      .select('patient_id patient_name overall_risk_score overall_risk_level')
  ]);

  return { riskDistribution, alertCounts, topHighRisk };
};

// Per-patient lab trend history — chart-ready
const getLabTrendChartData = async (patient_id) => {
  const trends = await LabTrend.find({ patient_id });
  return trends.map(t => ({

      value: p.value,
      status: p.status,
```

```js
const getPatientRiskProfile = async (req, res) => {
  try {
    const profile = await PatientRiskProfile.findOne({ patient_id: req.params.patientId });
    if (!profile) return res.status(404).json({ message: 'Risk profile not yet computed for this patient.' });
    res.json(profile);

// GET /predictive-care/profiles
const getAllRiskProfiles = async (req, res) => {
  try {
    const { risk_level, limit = 20, page = 1 } = req.query;
    const filter = risk_level ? { overall_risk_level: risk_level } : {};
    const profiles = await PatientRiskProfile.find(filter)
      .sort({ overall_risk_score: -1 })
      .skip((page - 1) * limit)
      .limit(parseInt(limit));
    const total = await PatientRiskProfile.countDocuments(filter);
    res.json({ data: profiles, total, page: parseInt(page), limit: parseInt(limit) });
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};

// POST /predictive-care/profiles/:patientId/compute
const computeForPatient = async (req, res) => {
  try {
    const patient = await Patient.findOne({ patient_id: req.params.patientId });
    if (!patient) return res.status(404).json({ message: 'Patient not found.' });

    await Promise.all([
      computeLabTrendsForPatient(patient.patient_id, `${patient.first_name} ${patient.last_name}`),
      computeRiskProfileForPatient(patient),
      checkVaccinationGapsForPatient(patient.patient_id, `${patient.first_name} ${patient.last_name}`),
      computeAdherenceForPatient(patient.patient_id, `${patient.first_name} ${patient.last_name}`),
    ]);

    const profile = await PatientRiskProfile.findOne({ patient_id: patient.patient_id });
    res.json({ message: 'Risk profile computed successfully.', profile });
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};

// POST /predictive-care/profiles/compute-all  (admin / cron trigger)
const computeForAll = async (req, res) => {
  try {
    computeAllPatientRiskProfiles(); // fire and forget
    res.json({ message: 'Computation started for all active patients.' });
  } catch (err) {
    res.status(500).json({ message: err.message });

};
```

```js
// src/api/v1/predictive-care/controllers/alerts.controller.js
const CareAlert = require('../models/careAlert.model');

// GET /predictive-care/alerts?patient_id=&alert_type=&severity=&is_resolved=
const getAlerts = async (req, res) => {
  try {
    const { patient_id, alert_type, severity, is_resolved, limit = 20, page = 1 } = req.query;
    const filter = {};
    if (patient_id) filter.patient_id = patient_id;
    if (alert_type) filter.alert_type = alert_type;
    if (severity) filter.severity = severity;
    if (is_resolved !== undefined) filter.is_resolved = is_resolved === 'true';

    const alerts = await CareAlert.find(filter)
      .sort({ triggered_at: -1 })
      .skip((page - 1) * limit)
      .limit(parseInt(limit));
    const total = await CareAlert.countDocuments(filter);
    res.json({ data: alerts, total, page: parseInt(page), limit: parseInt(limit) });
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};

// PATCH /predictive-care/alerts/:id/resolve
const resolveAlert = async (req, res) => {
  try {
    const alert = await CareAlert.findByIdAndUpdate(
      req.params.id,
      { is_resolved: true, resolved_at: new Date(), resolved_by: req.body.resolved_by || 'staff' },
      { new: true }
    );
    if (!alert) return res.status(404).json({ message: 'Alert not found.' });
    res.json(alert);
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};

// PATCH /predictive-care/alerts/:id/read
const markAlertRead = async (req, res) => {
  try {
    const alert = await CareAlert.findByIdAndUpdate(req.params.id, { is_read: true }, { new: true });
    if (!alert) return res.status(404).json({ message: 'Alert not found.' });
    res.json(alert);
  } catch (err) {
    res.status(500).json({ message: err.message });
  }
};
```

## Phase 4 — Validators

```js
// src/api/v1/predictive-care/validators/predictiveCare.validators.js
const { body, param, query } = require('express-validator');

const validatePatientIdParam = [
  param('patientId').notEmpty().withMessage('patientId is required'),
];

const validateAlertQuery = [
  query('severity').optional().isIn(['Info', 'Warning', 'Critical']).withMessage('Invalid severity'),
  query('alert_type').optional().isIn([
    'LAB_TREND', 'CHRONIC_RISK', 'VACCINATION_GAP',
    'ADHERENCE_GAP', 'NO_SHOW_RISK', 'READMISSION_RISK', 'CRITICAL_LAB'
  ]).withMessage('Invalid alert_type'),
  query('is_resolved').optional().isBoolean().withMessage('is_resolved must be boolean'),
  query('page').optional().isInt({ min: 1 }).withMessage('page must be a positive integer'),
  query('limit').optional().isInt({ min: 1, max: 100 }).withMessage('limit must be 1–100'),
];

const validateResolveAlert = [
  param('id').isMongoId().withMessage('Invalid alert ID'),
  body('resolved_by').optional().isString(),
];

const validateRiskProfileQuery = [
  query('risk_level').optional().isIn(['Low', 'Moderate', 'High', 'Critical']).withMessage('Invalid risk_level'),
  query('page').optional().isInt({ min: 1 }),
  query('limit').optional().isInt({ min: 1, max: 100 }),
];

module.exports = {
  validatePatientIdParam,
  validateAlertQuery,
  validateResolveAlert,
  validateRiskProfileQuery,
};
```
